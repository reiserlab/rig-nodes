cmake_minimum_required(VERSION 3.8)
project(bench_cpp)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(bench_msgs REQUIRED)
add_executable(sub_image src/sub_image.cpp)
ament_target_dependencies(sub_image rclcpp sensor_msgs)
add_executable(fixed src/fixed.cpp)
ament_target_dependencies(fixed rclcpp bench_msgs)
install(TARGETS sub_image fixed DESTINATION lib/${PROJECT_NAME})
ament_package()
